package "Embedding Generation Models" {

    class BERTEmbedder {
        - model: BertModel
        - tokenizer: BertTokenizer
        - max_length: int = 512
        - embedding_dim: int = 768
        --
        + generate_embeddings(texts: List[str]): ndarray
        + encode_batch(texts: List[str]): ndarray
        - preprocess(text: str): List[Token]
        + get_embedding_dimension(): int
    }

    class BoWEmbedder {
        - vectorizer: CountVectorizer
        - tfidf_transformer: TfidfTransformer
        - vocabulary_size: int
        - max_features: int = 5000
        --
        + fit_transform(texts: List[str]): ndarray
        + transform(texts: List[str]): ndarray
        + get_vocabulary(): Dict
        - build_vocabulary(texts): void
    }

    class Word2VecEmbedder {
        - model: Word2Vec
        - vector_size: int = 300
        - window: int = 5
        - min_count: int = 2
        --
        + train(texts: List[str]): void
        + generate_embeddings(texts: List[str]): ndarray
        + get_word_vector(word: str): ndarray
        - average_word_vectors(tokens): ndarray
    }
}

package "GAI Process Enhancement Layer" {
    class GAIProcessEnhancer {
        - api_key: str
        - model: str = "gpt-4"
        - provider: str = "openai"
        --
        + augment_dataset(texts: List[str], n_samples: int): List[str]
        + suggest_som_parameters(embeddings: ndarray): SOMConfig
        + interpret_cluster(docs: List[str]): ClusterInterpretation
        + generate_insights(results: ClusteringResults): str
        + analyze_quality_issues(metrics: Dict): str
    }

    class LocalLLM {
        - model: str = "claude-3-sonnet"
        - api_url: str
        --
        + generate(prompt: str): str
        + is_available(): bool
        + validate_synthetic_quality(text: str): float
    }
    
    note right of GAIProcessEnhancer
      GAI Role: Process Enhancement
      - Data augmentation
      - Parameter optimization
      - Cluster interpretation
      NOT document generation
    end note
}

package "SOM Clustering Engine" {
    class SOMTrainer {
        - grid_size: Tuple[int, int]
        - learning_rate: float
        - sigma: float
        - iterations: int
        - topology: str = "hexagonal"
        - weights: ndarray
        --
        + initialize(input_dim: int): void
        + train(data: ndarray): void
        + find_bmu(input_vector: ndarray): Tuple[int, int]
        + update_weights(bmu: Tuple, input_vector): void
        + decay_parameters(iteration: int): void
        + assign_clusters(data: ndarray): List[int]
        + get_u_matrix(): ndarray
    }

    class ClusterAssigner {
        - som: SOMTrainer
        - cluster_map: Dict
        --
        + assign_documents(embeddings: ndarray): List[int]
        + get_cluster_members(cluster_id: int): List[int]
        + calculate_cluster_centroids(): Dict
    }
}

package "Evaluation Module" {
    class ClusterEvaluator {
        - embeddings: ndarray
        - labels: List[int]
        --
        + calculate_silhouette(): float
        + calculate_davies_bouldin(): float
        + calculate_calinski_harabasz(): float
        + calculate_intra_cluster_distance(): float
        + calculate_inter_cluster_distance(): float
        + get_overall_quality_score(): float
    }

    class MetricsCollector {
        - results: List[ClusteringResult]
        --
        + add_result(result: ClusteringResult): void
        + compare_configurations(): DataFrame
        + get_best_configuration(): ClusteringResult
        + export_metrics(format: str): void
    }
}

package "Data Models" {
    class TextDocument {
        + id: str
        + text: str
        + preprocessed_text: str
        + cluster_id: int
        + embedding: ndarray
        --
        + to_dict(): Dict
        + get_tokens(): List[str]
    }

    class EmbeddingVector {
        + document_id: str
        + embedding_type: str
        + vector: ndarray
        + dimension: int
        --
        + normalize(): ndarray
        + compute_similarity(other: EmbeddingVector): float
    }

    class ClusteringResult {
        + config_id: str
        + embedding_type: str
        + som_config: SOMConfig
        + cluster_assignments: List[int]
        + silhouette_score: float
        + davies_bouldin_index: float
        + calinski_harabasz_score: float
        + training_time: float
        + quality_score: float
        --
        + to_dict(): Dict
        + is_high_quality(): bool
    }
    
    class SOMConfig {
        + grid_size: Tuple[int, int]
        + learning_rate: float
        + sigma: float
        + iterations: int
        + topology: str
        --
        + validate(): bool
        + to_dict(): Dict
    }
    
    class ClusterInterpretation {
        + cluster_id: int
        + label: str
        + summary: str
        + key_terms: List[str]
        + confidence: float
        + sample_documents: List[str]
        --
        + to_dict(): Dict
    }
}

package "Storage Layer" {
    class PostgreSQLDatabase {
        - connection: Connection
        - db_url: str
        --
        + store_embeddings(embeddings: List[EmbeddingVector]): void
        + store_clustering_result(result: ClusteringResult): str
        + store_interpretation(interpretation: ClusterInterpretation): void
        + query_results(filters: Dict): List[ClusteringResult]
        + get_best_results(): List[ClusteringResult]
    }

    class FileStorage {
        - base_path: str
        --
        + save_dataset(texts: List[str], filename: str): void
        + save_model(model: SOMTrainer, filename: str): void
        + save_visualization(plot: Figure, filename: str): void
        + load_dataset(filename: str): List[str]
    }
}

package "End-to-End Pipeline" {
    class TextClusteringPipeline {
        - bert_embedder: BERTEmbedder
        - bow_embedder: BoWEmbedder
        - w2v_embedder: Word2VecEmbedder
        - gai_enhancer: GAIProcessEnhancer
        - som_trainer: SOMTrainer
        - evaluator: ClusterEvaluator
        - database: PostgreSQLDatabase
        - preprocessor: TextPreprocessor
        --
        + process_dataset(texts: List[str]): ClusteringResults
        + process_with_augmentation(texts: List[str]): ClusteringResults
        - generate_all_embeddings(texts): Dict[str, ndarray]
        - train_som_grid_search(embeddings): List[ClusteringResult]
        + interpret_clusters_with_gai(): List[ClusterInterpretation]
        + export_results(format: str): void
    }

    class TextPreprocessor {
        - stop_words: Set[str]
        - lemmatizer: WordNetLemmatizer
        --
        + preprocess(text: str): str
        + clean_text(text: str): str
        + tokenize(text: str): List[str]
        + remove_stopwords(tokens: List[str]): List[str]
        + lemmatize(tokens: List[str]): List[str]
    }

    class DataValidator {
        - min_documents: int = 10
        - max_documents: int = 100000
        --
        + validate_dataset(texts: List[str]): ValidationReport
        + check_format(texts: List[str]): bool
        + analyze_balance(texts: List[str]): Dict
        + get_statistics(texts: List[str]): Dict
    }
}
