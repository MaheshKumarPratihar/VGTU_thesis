@startuml TextClustering_Component_EndToEnd
!theme cerulean-outline
title GAI-Enhanced Text Clustering with SOM - System Architecture

package "User Interface Layer" {
    [Web Dashboard\n(Streamlit/React)] as WebUI
    [REST API\n(Flask/FastAPI)] as API
    [Jupyter Interface] as Jupyter
}

package "Business Logic Layer" {
    component "Clustering Pipeline" as Pipeline {
        [Process Orchestrator] as Orch
        [Data Quality Gate] as DataQG
        [Cluster Quality Gate] as ClusterQG
    }
}

package "Data Processing Layer" {
    component "Preprocessing Module" {
        [Text Cleaner] as Cleaner
        [Tokenizer] as Tokenizer
        [Normalizer] as Normalizer
    }
    
    component "Embedding Module" {
        [BERT Encoder] as BERT
        [BoW Encoder] as BoW
        [Word2Vec Encoder] as W2V
    }
    
    component "Validation Module" {
        [Format Validator] as FormatVal
        [Quality Checker] as QualityCheck
    }
}

package "Clustering Layer" {
    component "SOM Engine" {
        [SOM Initializer] as SOMInit
        [BMU Finder] as BMU
        [Weight Updater] as WeightUpdate
        [Cluster Assigner] as Assigner
    }
    
    component "Evaluation Module" {
        [Silhouette Calculator] as Silhouette
        [Davies-Bouldin Calculator] as DB
        [Calinski-Harabasz Calculator] as CH
    }
}

package "GAI Enhancement Layer" {
    component "Process Enhancer" {
        [Data Augmenter] as Augment
        [Parameter Optimizer] as ParamOpt
        [Cluster Interpreter] as Interpreter
        [Insight Generator] as InsightGen
    }
}

package "Storage Layer" {
    database "PostgreSQL\nDatabase" as PostgreSQL {
        [Embeddings Cache] as EmbCache
        [Clustering Results] as ClusterResults
        [Configurations] as Configs
    }
    
    database "File Storage\n(Datasets & Models)" as FileStore {
        [Raw Datasets] as RawData
        [Trained SOM Models] as Models
        [Visualizations] as Viz
    }
}

package "Visualization Layer" {
    component "Plot Generator" {
        [SOM Heatmap] as Heatmap
        [t-SNE/UMAP] as TSNE
        [Silhouette Plot] as SilPlot
        [Metric Charts] as MetricChart
    }
}

package "External Services" {
    [OpenAI API\n(GPT-4)] as OpenAI
    [Anthropic API\n(Claude)] as Claude
    [Hugging Face\n(BERT Models)] as HuggingFace
}

' User Interface Connections
WebUI --> API
Jupyter --> API

' API to Pipeline
API --> Pipeline

' Pipeline to Data Processing
Pipeline --> Cleaner
Pipeline --> BERT
Pipeline --> BoW
Pipeline --> W2V
Pipeline --> FormatVal

' Pipeline to GAI Enhancement
Pipeline --> Augment
Pipeline --> ParamOpt
Pipeline --> Interpreter

' GAI to External Services
Augment --> OpenAI : cloud
Augment --> Claude : cloud
ParamOpt --> OpenAI : cloud
Interpreter --> Claude : cloud

' BERT to External Service
BERT --> HuggingFace : load models

' Pipeline to Clustering
Pipeline --> SOMInit
SOMInit --> BMU
BMU --> WeightUpdate
WeightUpdate --> Assigner

' Clustering to Evaluation
Assigner --> Silhouette
Assigner --> DB
Assigner --> CH

' Storage Connections
Pipeline --> PostgreSQL : store/retrieve
Pipeline --> FileStore : read/write

EmbCache <-- BERT : cache
EmbCache <-- BoW : cache
EmbCache <-- W2V : cache

ClusterResults <-- Assigner : store
Configs <-- ParamOpt : store

RawData <-- Cleaner : read
Models <-- SOMInit : save/load
Viz <-- Heatmap : save
Viz <-- TSNE : save

' Visualization
Pipeline --> Heatmap
Pipeline --> TSNE
Pipeline --> SilPlot
Pipeline --> MetricChart

' GAI Insight Generation
Interpreter --> InsightGen
InsightGen --> PostgreSQL : store insights

note right of Pipeline
  Core Output:
  Cluster assignments with
  quality metrics and
  GAI interpretations
end note

note right of Augment
  GAI Role #1:
  Data augmentation
  for balanced datasets
end note

note right of ParamOpt
  GAI Role #2:
  Hyperparameter optimization
  based on data analysis
end note

note right of Interpreter
  GAI Role #3:
  Cluster interpretation
  NOT document generation
end note

note bottom of PostgreSQL
  Primary Storage:
  - Embeddings (cached)
  - Clustering results
  - Quality metrics
  - GAI insights
end note

@enduml
