@startuml TextClustering_Sequence_EndToEnd
!theme cerulean-outline
title GAI-Enhanced Text Clustering with SOM & Multiple Embeddings - Sequence Diagram

actor "Researcher" as User
participant "System\nController" as System
participant "Data\nPreprocessor" as Preprocessor
participant "GAI\nService" as GAI
participant "BERT\nEncoder" as BERT
participant "BoW\nEncoder" as BoW
participant "Word2Vec\nEncoder" as W2V
participant "Hyperparameter\nOptimizer" as HyperOpt
participant "SOM\nTrainer" as SOM
participant "Evaluation\nEngine" as Eval
participant "Visualization\nModule" as Viz
database "Results\nDatabase" as DB

== Phase 1: Data Loading & Validation ==

User -> System: Upload text dataset
activate System

System -> Preprocessor: Validate dataset
activate Preprocessor

alt Dataset Invalid
    Preprocessor --> System: Validation errors
    System --> User: Display errors\n(format, encoding issues)
    User -> System: Upload corrected dataset
else Dataset Valid
    Preprocessor -> Preprocessor: Analyze statistics\n(doc count, vocab size)
    Preprocessor --> System: Dataset validated
end

deactivate Preprocessor

== Phase 2: Text Preprocessing ==

System -> Preprocessor: Preprocess texts
activate Preprocessor

Preprocessor -> Preprocessor: Clean HTML & URLs
Preprocessor -> Preprocessor: Tokenize & normalize
Preprocessor -> Preprocessor: Remove stopwords
Preprocessor -> Preprocessor: Apply lemmatization

Preprocessor --> System: Preprocessed data
deactivate Preprocessor

== Phase 3: GAI Data Augmentation ==

System -> GAI: Analyze class distribution
activate GAI

GAI -> GAI: Calculate class imbalance

alt Dataset Imbalanced
    GAI -> GAI: Identify underrepresented classes
    GAI -> GAI: Generate synthetic samples\n(contextually relevant)
    
    GAI --> System: Synthetic samples
    
    System -> Preprocessor: Validate synthetic quality
    activate Preprocessor
    
    Preprocessor -> Preprocessor: Check coherence & relevance
    
    alt Quality >= 0.85
        Preprocessor --> System: Quality approved
        System -> System: Merge with original dataset
    else Quality < 0.85
        Preprocessor --> System: Quality too low
        System -> GAI: Regenerate samples
        GAI -> GAI: Improve generation
        GAI --> System: New synthetic samples
    end
    
    deactivate Preprocessor
else Dataset Balanced
    GAI --> System: No augmentation needed
end

deactivate GAI

== Phase 4: Parallel Embedding Generation ==

User -> System: Select embedding models (BERT, BoW, Word2Vec)

par BERT Embedding
    System -> BERT: Generate BERT embeddings
    activate BERT
    
    loop For each document batch
        BERT -> BERT: Tokenize with BERT tokenizer
        BERT -> BERT: Add [CLS] & [SEP] tokens
        BERT -> BERT: Forward pass
        BERT -> BERT: Extract [CLS] embeddings
    end
    
    BERT -> BERT: Normalize embeddings [n×768]
    BERT --> System: BERT embeddings ready
    deactivate BERT

else BoW Embedding
    System -> BoW: Generate BoW vectors
    activate BoW

    BoW -> BoW: Build vocabulary
    BoW -> BoW: Create term-document matrix
    BoW -> BoW: Apply TF-IDF
    BoW -> BoW: Normalize vectors [n×vocab_size]

    BoW --> System: BoW embeddings ready
    deactivate BoW

else Word2Vec Embedding
    System -> W2V: Generate Word2Vec embeddings
    activate W2V

    W2V -> W2V: Load/train Word2Vec model
    
    loop For each document
        W2V -> W2V: Get word vectors
        W2V -> W2V: Average vectors
    end
    
    W2V -> W2V: Normalize embeddings [n×300]
    W2V --> System: Word2Vec embeddings ready
    deactivate W2V
end

System -> DB: Store all embeddings
activate DB
DB --> System: Storage confirmed
deactivate DB


== Phase 5: GAI Hyperparameter Optimization ==

System -> GAI: Request parameter suggestions
activate GAI

GAI -> GAI: Analyze embedding dimensions
GAI -> GAI: Calculate data density
GAI -> GAI: Assess complexity

GAI -> HyperOpt: Suggest parameter ranges
activate HyperOpt

HyperOpt -> HyperOpt: Generate grid:\n- Grid size: [(10,10), (15,15), (20,20)]\n- Learning rate: [0.1, 0.5, 1.0]\n- Sigma: [1.0, 2.0, 3.0]\n- Iterations: [1000, 5000, 10000]

HyperOpt --> GAI: Parameter grid created
deactivate HyperOpt

GAI --> System: Parameter suggestions
deactivate GAI

User -> System: Review & approve parameters
System -> System: Create final hyperparameter grid

== Phase 6: SOM Training Loop ==

loop For each embedding type (BERT, BoW, Word2Vec)
    
    loop For each hyperparameter configuration
        
        System -> SOM: Initialize SOM with config
        activate SOM
        
        SOM -> SOM: Create grid topology
        SOM -> SOM: Initialize random weights
        
        SOM -> SOM: Train SOM
        note right
          Training Process:
          1. Select random input
          2. Find Best Matching Unit (BMU)
          3. Calculate neighborhood
          4. Update BMU & neighbors
          5. Decay learning rate & sigma
        end note
        
        SOM -> SOM: Assign documents to clusters
        SOM -> SOM: Calculate cluster statistics
        
        SOM --> System: Clustering results\n(cluster assignments, centroids)
        
        deactivate SOM
        
        System -> Eval: Evaluate clustering quality
        activate Eval
        
        Eval -> Eval: Calculate Silhouette score
        Eval -> Eval: Calculate Davies-Bouldin index
        Eval -> Eval: Calculate Calinski-Harabasz score
        Eval -> Eval: Calculate intra-cluster distance
        Eval -> Eval: Calculate inter-cluster distance
        
        Eval --> System: Quality metrics
        deactivate Eval
        
        System -> DB: Store configuration & results
        activate DB
        DB --> System: Stored successfully
        deactivate DB
        
    end
    
    System -> System: Identify best config\nfor current embedding
    
end

== Phase 7: Quality Validation ==

System -> Eval: Compare all configurations
activate Eval

Eval -> Eval: Rank by performance metrics
Eval -> Eval: Calculate overall quality score

alt Quality Score >= 0.70
    Eval --> System: Results acceptable
else Quality Score < 0.70
    Eval --> System: Poor quality detected
    
    System -> GAI: Analyze performance issues
    activate GAI
    
    GAI -> GAI: Identify problems:\n- Too many/few clusters\n- Poor separation\n- Unbalanced clusters
    
    GAI --> System: Suggestions for improvement
    deactivate GAI
    
    alt Accept suggestions
        System -> SOM: Re-train with new parameters
        note right: Return to Phase 6
    else Reject
        System -> User: Flag for manual review
    end
end

deactivate Eval

== Phase 8: GAI Cluster Interpretation ==

System -> GAI: Request cluster interpretation
activate GAI

loop For each cluster
    
    System -> GAI: Send cluster documents (sample)
    
    GAI -> GAI: Analyze document content
    GAI -> GAI: Identify common themes
    GAI -> GAI: Generate semantic label
    GAI -> GAI: Create cluster summary
    GAI -> GAI: Extract key terms
    
    alt Interpretation Confidence >= 0.80
        GAI --> System: Cluster metadata\n(label, summary, keywords)
        
        System -> DB: Store cluster interpretation
        activate DB
        DB --> System: Stored successfully
        deactivate DB
    else Confidence < 0.80
        GAI -> GAI: Request more context
        System -> GAI: Send additional samples
        GAI -> GAI: Re-analyze with more data
        GAI --> System: Improved interpretation
    end
    
end

deactivate GAI

== Phase 9: Comparative Analysis ==

System -> GAI: Generate comparative insights
activate GAI

GAI -> GAI: Compare embedding methods
GAI -> GAI: Analyze trade-offs
GAI -> GAI: Generate recommendations

GAI --> System: Comparative analysis report
deactivate GAI

== Phase 10: Visualization & Reporting ==

System -> Viz: Generate visualizations
activate Viz

par Create SOM U-Matrix heatmap
    Viz -> Viz: Create SOM U-Matrix heatmap

else Generate t-SNE/UMAP 2D plot
    Viz -> Viz: Generate t-SNE/UMAP 2D plot

else Plot Silhouette diagram
    Viz -> Viz: Plot Silhouette diagram

else Create metric comparison charts
    Viz -> Viz: Create metric comparison charts

else Generate cluster size distribution
    Viz -> Viz: Generate cluster size distribution
end

Viz --> System: All visualizations ready
deactivate Viz

System -> GAI: Generate executive report
activate GAI

GAI -> GAI: Compile findings
GAI -> GAI: Create insights summary
GAI -> GAI: Generate recommendations

GAI --> System: Executive report
deactivate GAI

== Phase 11: Result Delivery ==

System -> DB: Retrieve all results
activate DB

DB --> System: "Complete result set:\n- Best configurations\n- All metrics\n- Cluster assignments\n- Interpretations\n- Visualizations"
deactivate DB

System --> User: Deliver comprehensive report
note right of User
  PRIMARY OUTPUT:
  - Optimal cluster assignments
  - Best embedding method
  - Quality metrics comparison
  - GAI-interpreted clusters
  - Visualizations
  - Actionable insights
end note

User -> User: Review results

alt Results Satisfactory
    User -> System: Export final deliverables
    System -> DB: Export data (CSV/JSON/PDF)
    activate DB
    DB --> System: Export complete
    deactivate DB
    System --> User: Files ready for download

else Results Need Refinement
    User -> System: Request re-training\nwith different parameters
    note right of System: Return to Phase 5 or 6
end

@enduml
